# 📋 Contoh Implementasi Modul rakitin

## 🇮🇩 Bahasa Indonesia

### Pendahuluan

Dokumen ini menyediakan contoh implementasi modul yang dihasilkan oleh **rakitin CLI**. Contoh-contoh ini mencakup berbagai arsitektur dan integrasi ORM yang didukung oleh rakitin.

### Struktur Modul

#### 1. Arsitektur Simple

Dalam arsitektur simple, semua komponen modul (controller, service, router) berada dalam satu direktori modul.

```
app/modules/user/
├── user.controller.js
├── user.service.js
└── user.routes.js
```

#### 2. Arsitektur Modular

Dalam arsitektur modular, komponen modul dipisahkan ke dalam sub-direktori.

```
app/modules/user/
├── controllers/
│   └── user.controller.js
├── services/
│   └── user.service.js
├── models/
│   └── user.model.js
└── routes/
    └── user.routes.js
```

```javascript
// Generated by rakitin
// Service untuk Product

const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

class ProductService {
  /**
   * Mendapatkan semua product
   */
  static async getAll() {
    try {
      const products = await prisma.product.findMany({
        include: {
          category: true
        }
      });
      
      return products;
    } catch (error) {
      throw error;
    }
  }

  /**
   * Mendapatkan product by ID
   */
  static async getById(id) {
    try {
      const product = await prisma.product.findUnique({
        where: { id: parseInt(id) },
        include: {
          category: true
        }
      });
      
      if (!product) {
        throw new Error('Product not found');
      }
      
      return product;
    } catch (error) {
      throw error;
    }
  }

  /**
   * Membuat product baru
   */
  static async create(productData) {
    try {
      const { name, description, price, categoryId, stock } = productData;
      
      const product = await prisma.product.create({
        data: {
          name,
          description,
          price: parseFloat(price),
          categoryId: parseInt(categoryId),
          stock: parseInt(stock)
        },
        include: {
          category: true
        }
      });
      
      return product;
    } catch (error) {
      throw error;
    }
  }

  /**
   * Memperbarui product
   */
  static async update(id, productData) {
    try {
      const { name, description, price, categoryId, stock } = productData;
      
      const product = await prisma.product.update({
        where: { id: parseInt(id) },
        data: {
          name,
          description,
          price: price ? parseFloat(price) : undefined,
          categoryId: categoryId ? parseInt(categoryId) : undefined,
          stock: stock ? parseInt(stock) : undefined
        },
        include: {
          category: true
        }
      });
      
      return product;
    } catch (error) {
      throw error;
    }
  }

  /**
   * Menghapus product
   */
  static async delete(id) {
    try {
      await prisma.product.delete({
        where: { id: parseInt(id) }
      });
      
      return { success: true };
    } catch (error) {
      throw error;
    }
  }
}

module.exports = ProductService;
```

##### services/order.service.js

```javascript
// Generated by rakitin
// Service untuk Order

const { Order, OrderItem, Product } = require('../models');

class OrderService {
  /**
   * Mendapatkan semua order
   */
  static async getAll() {
    try {
      const orders = await Order.findAll({
        include: [
          {
            model: OrderItem,
            include: [Product]
          }
        ]
      });
      
      return orders;
    } catch (error) {
      throw error;
    }
  }

  /**
   * Mendapatkan order by ID
   */
  static async getById(id) {
    try {
      const order = await Order.findByPk(id, {
        include: [
          {
            model: OrderItem,
            include: [Product]
          }
        ]
      });
      
      if (!order) {
        throw new Error('Order not found');
      }
      
      return order;
    } catch (error) {
      throw error;
    }
  }

  /**
   * Membuat order baru
   */
  static async create(orderData) {
    try {
      const { userId, items, shippingAddress } = orderData;
      
      // Hitung total amount
      let totalAmount = 0;
      for (const item of items) {
        const product = await Product.findByPk(item.productId);
        if (!product) {
          throw new Error(`Product with ID ${item.productId} not found`);
        }
        totalAmount += product.price * item.quantity;
      }
      
      // Buat order
      const order = await Order.create({
        userId,
        totalAmount,
        status: 'PENDING',
        shippingAddress
      });
      
      // Buat order items
      for (const item of items) {
        await OrderItem.create({
          orderId: order.id,
          productId: item.productId,
          quantity: item.quantity,
          price: item.price
        });
      }
      
      // Return order dengan items
      return await Order.findByPk(order.id, {
        include: [
          {
            model: OrderItem,
            include: [Product]
          }
        ]
      });
    } catch (error) {
      throw error;
    }
  }

  /**
   * Memperbarui status order
   */
  static async updateStatus(id, status) {
    try {
      const order = await Order.findByPk(id);
      
      if (!order) {
        throw new Error('Order not found');
      }
      
      order.status = status;
      await order.save();
      
      return await Order.findByPk(id, {
        include: [
          {
            model: OrderItem,
            include: [Product]
          }
        ]
      });
    } catch (error) {
      throw error;
    }
  }

  /**
   * Menghapus order
   */
  static async delete(id) {
    try {
      const order = await Order.findByPk(id);
      
      if (!order) {
        throw new Error('Order not found');
      }
      
      // Hapus order items terlebih dahulu
      await OrderItem.destroy({
        where: { orderId: id }
      });
      
      // Hapus order
      await order.destroy();
      
      return { success: true };
    } catch (error) {
      throw error;
    }
  }
}

module.exports = OrderService;
```

### Integrasi dengan Router Utama

Setelah membuat modul, Anda dapat mengintegrasikannya dengan router utama menggunakan fitur "Integrasi Router Utama" dari rakitin.

#### Contoh Hasil Integrasi Router Utama

```javascript
// Generated by rakitin
// Main router

const express = require('express');
const router = express.Router();

// Import module routers
const userRouter = require('../modules/user/user.routes');
const productRouter = require('../modules/product/routes/product.routes');
const orderRouter = require('../modules/order/routes/order.routes');

// Use module routers
router.use('/user', userRouter);
router.use('/product', productRouter);
router.use('/order', orderRouter);

module.exports = router;
```

### Penggunaan di app.js

Setelah integrasi router utama, Anda dapat menggunakan router tersebut di file app.js:

```javascript
const express = require('express');
const app = express();

// Middleware
app.use(express.json());

// Use main router
const routes = require('./routes');
app.use('/api', routes);

module.exports = app;
```

```
app/modules/user/
├── user.controller.js
```

In modular architecture, module components are separated into sub-directories.

```
app/modules/user/
├── controllers/
│   └── user.controller.js
├── services/
│   └── user.service.js
├── models/
│   └── user.model.js
```


##### services/order.service.js

```javascript
// Generated by rakitin
// Service for Order

const { Order, OrderItem, Product } = require('../models');

class OrderService {
  /**
   * Get all orders
   */
  static async getAll() {
    try {
      const orders = await Order.findAll({
        include: [
          {
            model: OrderItem,
            include: [Product]
          }
        ]
      });
      
      return orders;
    } catch (error) {
      throw error;
    }
  }

  /**
   * Get order by ID
   */
  static async getById(id) {
    try {
      const order = await Order.findByPk(id, {
        include: [
          {
            model: OrderItem,
            include: [Product]
          }
        ]
      });
      
      if (!order) {
        throw new Error('Order not found');
      }
      
      return order;
    } catch (error) {
      throw error;
    }
  }

  /**
   * Create new order
   */
  static async create(orderData) {
    try {
      const { userId, items, shippingAddress } = orderData;
      
      // Calculate total amount
      let totalAmount = 0;
      for (const item of items) {
        const product = await Product.findByPk(item.productId);
        if (!product) {
          throw new Error(`Product with ID ${item.productId} not found`);
        }
        totalAmount += product.price * item.quantity;
      }
      
      // Create order
      const order = await Order.create({
        userId,
        totalAmount,
        status: 'PENDING',
        shippingAddress
      });
      
      // Create order items
      for (const item of items) {
        await OrderItem.create({
          orderId: order.id,
          productId: item.productId,
          quantity: item.quantity,
          price: item.price
        });
      }
      
      // Return order with items
      return await Order.findByPk(order.id, {
        include: [
          {
            model: OrderItem,
            include: [Product]
          }
        ]
      });
    } catch (error) {
      throw error;
    }
  }

  /**
   * Update order status
   */
  static async updateStatus(id, status) {
    try {
      const order = await Order.findByPk(id);
      
      if (!order) {
        throw new Error('Order not found');
      }
      
      order.status = status;
      await order.save();
      
      return await Order.findByPk(id, {
        include: [
          {
            model: OrderItem,
            include: [Product]
          }
        ]
      });
    } catch (error) {
      throw error;
    }
  }

  /**
   * Delete order
   */
  static async delete(id) {
    try {
      const order = await Order.findByPk(id);
      
      if (!order) {
        throw new Error('Order not found');
      }
      
      // Delete order items first
      await OrderItem.destroy({
        where: { orderId: id }
      });
      
      // Delete order
      await order.destroy();
      
      return { success: true };
    } catch (error) {
      throw error;
    }
  }
}

module.exports = OrderService;
```
